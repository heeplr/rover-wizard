43def baud_rates_to_try(ctx, port: int, preferred=()) -> list: 

44 """SERIALx_BAUD values for a probe: the port's current rate, then the device's usual ones, then common rates.""" 

45 current = ctx.param_int(f"SERIAL{port}_BAUD", DEFAULT_SERIAL_BAUD) or DEFAULT_SERIAL_BAUD 

46 ordered = [] 

47 for baud in (current, *preferred, *COMMON_PROBE_BAUDS): 

48 if baud not in ordered: 

49 ordered.append(baud) 

50 return ordered
