76def selected_source_set(ctx) -> int: 

77 """The EKF source set (1..3) the step's fields currently describe."""

96def action_use_source_set(ctx): 

97 """Switch the EKF to the selected source set now (apply its settings first).""" 

98 source_set = selected_source_set(ctx) 

99 result = ctx.vehicle.set_ekf_source_set(source_set) 

100 if result not in ACCEPTED_RESULTS: 100 ↛ 101line 100 didn't jump to line 101 because the condition on line 100 was never true

101 raise VehicleError(f"Source set {source_set} refused (result {result}). {ctx.vehicle.last_text()}") 

102 ctx.log(f"EKF now uses source set {source_set}.")

105def has_gps_heading(ctx) -> bool: 

106 """A receiver that measures heading (dual antenna / moving baseline) is configured.""" 

107 return any(ctx.param_int(names) in GPS_TYPES_WITH_HEADING for names in GPS_TYPE_PARAMETERS)

110def proposed_heading_source(ctx): 

111 """Heading source to propose for source set 1, with the reason - or (None, "") if the current one fits.""" 

112 current = ctx.param_int("EK3_SRC1_YAW", EKF_SOURCE_COMPASS) 

113 if current != EKF_SOURCE_COMPASS: 113 ↛ 114line 113 didn't jump to line 114 because the condition on line 113 was never true

114 return None, "" 

115 if has_gps_heading(ctx): 

116 return EKF_YAW_GPS_COMPASS_FALLBACK, "a GPS that measures heading is configured" 

117 if not has("has_compass")(ctx) and has("has_gps")(ctx): 

118 return EKF_YAW_GSF, "the vehicle has no compass" 

119 return None, ""

122def propose_heading_source(ctx): 

123 """When the step opens: propose a better heading source for set 1 (written only by Apply).""" 

124 proposal, reason = proposed_heading_source(ctx) 

125 if proposal is None or selected_source_set(ctx) != 1 or ctx.values.get("src_yaw") != EKF_SOURCE_COMPASS: 

126 return 

127 ctx.set_values(src_yaw=proposal) 

128 label = dict((value, name) for name, value in EKF_YAW_CHOICES)[proposal] 

129 ctx.log(f"Heading from: proposing '{label}' because {reason} - press Apply to use it.")

137def imu_bit_field(number: int): 

138 """EK3_IMU_MASK checkbox for IMU `number`, shown when ArduPilot detected that IMU.""" 

139 id_parameter = "INS_ACC_ID" if number == 1 else f"INS_ACC{number}_ID" 

140 return bit_field( 

141 "EK3_IMU_MASK", 

142 1 << (number - 1), 

143 f"EKF instance for IMU {number}", 

144 key=f"ekf_imu{number}", 

145 when=lambda ctx: bool(ctx.param_int(id_parameter)), 

146 )
