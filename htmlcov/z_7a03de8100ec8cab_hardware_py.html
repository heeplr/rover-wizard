38def _are_relays_configured(ctx) -> bool: 

39 """Detector: a relay is enabled, switched by RC, or an output is set up as a GPIO pin.""" 

40 relay_enabled = any(ctx.param_int(f"RELAY{relay}_FUNCTION") for relay in RELAYS) 

41 relay_switch = any(ctx.param_int(f"RC{channel}_OPTION") in RC_OPTION_RELAY.values() for channel in RC_CHANNELS) 

42 return relay_enabled or relay_switch or _does_any_output_have_function(ctx, {SERVO_FUNCTION_GPIO})

92def _is_gps_fitted(ctx) -> bool: 

93 """A GPS type is set and - once GPS telemetry has arrived - a receiver is actually connected. 

94 

95 ArduPilot's default GPS type is Auto, so the type alone doesn't prove there is a receiver; the fix type in 

96 GPS_RAW_INT / GPS2_RAW is "no GPS connected" (0) when none answers. 

97 """ 

98 if not _any_parameter_nonzero(GPS_TYPE_PARAMETERS)(ctx): 

99 return False 

100 reports = [report for report in (ctx.vehicle.latest("GPS_RAW_INT"), ctx.vehicle.latest("GPS2_RAW")) if report] 

101 return not reports or any(report.fix_type >= GPS_FIX_TYPE_NO_FIX for report in reports) 

102

104def _is_compass_fitted(ctx) -> bool: 

105 """Compasses are enabled and ArduPilot has detected one: it records each detected compass's device id. 

106 

107 COMPASS_ENABLE alone isn't enough - it defaults to 1 with or without a compass. 

108 """ 

109 if not ctx.param_int("COMPASS_ENABLE", 1): 

110 return False 

111 known_ids = [name for name in COMPASS_DEVICE_ID_PARAMETERS if ctx.vehicle.has_parameter(name)] 

112 # Without device-id parameters (unknown firmware) the enable switch is all there is to go by. 

113 return not known_ids or _any_parameter_nonzero(known_ids)(ctx)

123def _is_sensor_detected(ctx, id_parameters) -> bool: 

124 """A detected sensor has a device id; without those parameters (unknown firmware), IMU telemetry decides.""" 

125 known_ids = [name for name in id_parameters if ctx.vehicle.has_parameter(name)] 

126 if known_ids: 

127 return _any_parameter_nonzero(known_ids)(ctx) 

128 return ctx.vehicle.imu() is not None

141def _is_rc_receiver_fitted(ctx) -> bool: 

142 """The receiver delivers channels (RC_CHANNELS.chancount > 0); there is no parameter for this. 

143 

144 Before any RC telemetry has arrived the answer is unknown: assume a receiver, so the RC step stays visible. 

145 """ 

146 raw = ctx.vehicle.rc_raw() 

147 return True if raw is None else bool(raw)
