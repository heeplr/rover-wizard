1"""rover_wizard.vehicle.boat_telemetry - what a boat or sailing vehicle reports besides the usual telemetry: the wind 

2(WIND and two NAMED_VALUE_FLOATs, AP_WindVane::send_wind) and the water depth (WATER_DEPTH, boats only: 

3GCS_MAVLINK_Rover::send_water_depth). All of them come with the telemetry group Vehicle.setup_streams() requests.""" 

4 

5from __future__ import annotations

7from typing import Optional

9from ..boat_defs import APPARENT_WIND_DIRECTION_NAME, APPARENT_WIND_SPEED_NAME

12# [s] the wind and the depth arrive with the 2 Hz telemetry group: a few lost messages are no gap yet 

13BOAT_TELEMETRY_MAX_AGE_SECONDS = 2 * MONITOR_MAX_AGE_SECONDS

16def true_wind(vehicle) -> Optional[tuple]: 

17 """(direction the true wind comes from [deg, 0 = north], speed [m/s]), or None without fresh WIND messages.""" 

18 message = vehicle.latest("WIND", max_age=BOAT_TELEMETRY_MAX_AGE_SECONDS) 

19 return None if message is None else (message.direction, message.speed)

21 

22def apparent_wind(vehicle) -> Optional[tuple]: 

23 """(direction relative to the bow [deg, -180..180, + = from starboard], speed [m/s]) of the wind the vane feels, 

24 or None without fresh reports.""" 

25 direction = vehicle.latest(named_value_key(APPARENT_WIND_DIRECTION_NAME), max_age=BOAT_TELEMETRY_MAX_AGE_SECONDS) 

26 speed = vehicle.latest(named_value_key(APPARENT_WIND_SPEED_NAME), max_age=BOAT_TELEMETRY_MAX_AGE_SECONDS) 

27 return None if direction is None or speed is None else (direction.value, speed.value) 

28 

29 

30def water_depth(vehicle) -> Optional[tuple]: 

31 """(depth below the depth sounder [m], water temperature [deg C], healthy) from fresh WATER_DEPTH, or None.""" 

32 message = vehicle.latest("WATER_DEPTH", max_age=BOAT_TELEMETRY_MAX_AGE_SECONDS) 

33 return None if message is None else (message.distance, message.temperature, bool(message.healthy))
