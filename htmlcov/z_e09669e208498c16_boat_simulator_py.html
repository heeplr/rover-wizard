1"""rover_wizard.vehicle.boat_simulator - a simulated boat (FRAME_CLASS Boat) for --demo-boat and the tests. 

2 

3Besides the rover's hidden truths it hides a Torqeedo motor and an echosounder the wizard has to set up: the 

4Torqeedo only reports (ESC telemetry) once its type and its serial port (protocol, 19200 baud) are right, and the 

5depth only arrives (WATER_DEPTH, as Rover sends it for boats) from the hidden rangefinder pointing down. Like 

6ArduPilot, the Torqeedo's settings beyond its type appear only after a reboot with the type set. 

7""" 

8 

9from __future__ import annotations 

10 

11from types import SimpleNamespace 

12 

13from ..rover_defs import FRAME_CLASS_BOAT, RANGEFINDER_ORIENT_DOWN, SERVO_FUNCTION_THROTTLE 

14from ..boat_defs import MISSION_DONE_LOITER, SERIAL_PROTOCOL_TORQEEDO, TORQEEDO_SERIAL_BAUD 

15from ..mavlink_defs import MAV_PARAM_TYPE_REAL32, MAV_TYPE_SURFACE_BOAT, PARAM_INDEX_UNKNOWN 

16from .rover_simulator import SimVehicle 

17 

18# ESC_TELEMETRY_1_TO_4 carries four motors; the Torqeedo reports on its output's index 

19ESC_TELEMETRY_SLOTS = 4 

20CENTI = 100 

21 

22 

23class SimBoat(SimVehicle): 

24 """A simulated boat; see the module docstring.""" 

25 

26 # [m] water below the simulated echosounder, and its temperature [deg C] 

27 TRUE_DEPTH = 12.5 

28 WATER_TEMPERATURE = 14.0 

29 # what the simulated Torqeedo reports 

30 TORQEEDO_RPM = 850 

31 TORQEEDO_VOLTS = 29.4 

32 TORQEEDO_AMPS = 12.5 

33 TORQEEDO_TEMPERATURE = 31 

34 # firmware before 4.7 names the (only) Torqeedo's settings TRQD_* 

35 TORQEEDO_PREFIX = "TRQD_" 

36 

37 def __init__(self, skid: bool = False, wired_wrong: bool = False) -> None: 

38 """A boat with no Torqeedo and no echosounder fitted (tests set true_torqeedo / true_rangefinder).""" 

39 # (TRQD_TYPE, serial port) of a Torqeedo that answers when set up like that (None = not fitted) 

40 self.true_torqeedo = None 

41 super().__init__(skid=skid, wired_wrong=wired_wrong) 

42 self.vehicle_type = MAV_TYPE_SURFACE_BOAT 

43 

44 def _build_params(self) -> None: 

45 """The rover's parameters, with Rover's boat defaults (Rover::init_ardupilot) and the boat-only ones.""" 

46 super()._build_params() 

47 self._params["FRAME_CLASS"] = float(FRAME_CLASS_BOAT) 

48 self._add_params( 

49 { 

50 "LOIT_TYPE": 0, 

51 "LOIT_RADIUS": 2, 

52 "LOIT_SPEED_GAIN": 0.5, 

53 "MIS_DONE_BEHAVE": MISSION_DONE_LOITER, 

54 "MOT_VEC_ANGLEMAX": 0, 

55 # AP_GPS's default: Airborne 4G 

56 "GPS_NAVFILTER": 8, 

57 f"{self.TORQEEDO_PREFIX}TYPE": 0, 

58 } 

59 ) 

60 

61 def reboot(self, progress=None) -> None: 

62 """Like ArduPilot: the Torqeedo's settings exist only after a reboot with its type set.""" 

63 if self.get_int(f"{self.TORQEEDO_PREFIX}TYPE") and not self.has_parameter(f"{self.TORQEEDO_PREFIX}SLEW_TIME"): 

64 extra = {"ONOFF_PIN": -1, "DE_PIN": -1, "OPTIONS": 1, "POWER": 100, "SLEW_TIME": 2.0, "DIR_DELAY": 1.0} 

65 for suffix, value in extra.items(): 

66 self._store_param( 

67 self.TORQEEDO_PREFIX + suffix, 

68 float(value), 

69 MAV_PARAM_TYPE_REAL32, 

70 PARAM_INDEX_UNKNOWN, 

71 self._param_total, 

72 ) 

73 super().reboot(progress) 

74 

75 def _publish_telemetry(self, timestamp: float) -> None: 

76 """The rover's telemetry, plus the depth and the Torqeedo's reports when they are set up right.""" 

77 super()._publish_telemetry(timestamp) 

78 instance = self._answering_rangefinder() 

79 if ( 

80 instance is not None 

81 and self.get_int("FRAME_CLASS") == FRAME_CLASS_BOAT 

82 and self.get_int(f"RNGFND{instance}_ORIENT") == RANGEFINDER_ORIENT_DOWN 

83 ): 

84 self._store_message( 

85 "WATER_DEPTH", 

86 SimpleNamespace(distance=self.TRUE_DEPTH, temperature=self.WATER_TEMPERATURE, healthy=1), 

87 timestamp, 

88 ) 

89 if self._is_torqeedo_answering(): 

90 self._store_message("ESC_TELEMETRY_1_TO_4", self._torqeedo_telemetry(), timestamp) 

91 

92 def _is_torqeedo_answering(self) -> bool: 

93 """The hidden Torqeedo is fitted, its type is right and its serial port talks the Torqeedo protocol.""" 

94 if self.true_torqeedo is None: 

95 return False 

96 kind, port = self.true_torqeedo 

97 return self.get_int(f"{self.TORQEEDO_PREFIX}TYPE") == kind and self._is_serial_port_configured( 

98 port, SERIAL_PROTOCOL_TORQEEDO, TORQEEDO_SERIAL_BAUD 

99 ) 

100 

101 def _torqeedo_telemetry(self) -> SimpleNamespace: 

102 """ESC_TELEMETRY_1_TO_4 with the Torqeedo on its throttle output's slot.""" 

103 slot = ((self._output_with_function(SERVO_FUNCTION_THROTTLE) or 1) - 1) % ESC_TELEMETRY_SLOTS 

104 

105 def values(value): 

106 """The value in the Torqeedo's slot, 0 elsewhere.""" 

107 return [value if index == slot else 0 for index in range(ESC_TELEMETRY_SLOTS)] 

108 

109 return SimpleNamespace( 

110 rpm=values(self.TORQEEDO_RPM), 

111 voltage=values(round(self.TORQEEDO_VOLTS * CENTI)), 

112 current=values(round(self.TORQEEDO_AMPS * CENTI)), 

113 temperature=values(self.TORQEEDO_TEMPERATURE), 

114 )
