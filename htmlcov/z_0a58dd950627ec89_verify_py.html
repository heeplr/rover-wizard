98def default_speed_step(ctx) -> float: 

99 """Speed-step target: a share of the cruise speed, never below the minimum nor above Guided mode's limit.""" 

100 cruise_speed = ctx.param_value("CRUISE_SPEED", 0) or FALLBACK_CRUISE_SPEED 

101 # Share of the cruise speed, to 0.1 m/s, but not below the minimum step ... 

102 step = max(MIN_SPEED_STEP, round(SPEED_STEP_FRACTION * cruise_speed, 1)) 

103 limit = guided_speed_limit(ctx) 

104 # ... and not above WP_SPEED (to 0.1 m/s, rounded down so it stays within the limit). 

105 return min(step, math.floor(limit * 10) / 10) if limit else step

114def _manual_tracking_report(ctx): 

115 """Pilot drives in Acro; report how well achieved turn rate/speed followed the sticks (ratio and RMS error).""" 

116 samples = ctx.drive.record_manual( 

117 "Drive normally in ACRO mode (steering stick = turn rate, throttle = speed) with " 

118 "varied inputs for 20-30 s. Press Continue when done.", 

119 mode="ACRO", 

120 ) 

121 

122 # Full stick in Acro means ACRO_TURN_RATE (turning) and SPEED_MAX, or CRUISE_SPEED if unset (speed). 

123 full_turn_rate = math.radians(ctx.param_value("ACRO_TURN_RATE", 0) or FALLBACK_ACRO_TURN_RATE) 

124 full_speed = ctx.param_value("SPEED_MAX", 0) or (ctx.param_value("CRUISE_SPEED", 0) or FALLBACK_CRUISE_SPEED) 

125 for name, stick_attribute, measured_attribute, full_scale in ( 

126 ("Steering rate", "steering_input", "yaw_rate", full_turn_rate), 

127 ("Speed", "throttle_input", "forward_speed", full_speed), 

128 ): 

129 # Desired value = stick position x full-scale value; pair it with what the vehicle actually did. 

130 pairs = [ 

131 (getattr(sample, stick_attribute) * full_scale, getattr(sample, measured_attribute)) 

132 for sample in samples 

133 if getattr(sample, stick_attribute) is not None 

134 ] 

135 # Ignore small stick inputs: mostly noise and dead-band. 

136 pairs = [ 

137 (desired, achieved) 

138 for desired, achieved in pairs 

139 if abs(desired) > TRACKING_MIN_INPUT_FRACTION * full_scale 

140 ] 

141 if len(pairs) < TRACKING_MIN_PAIRS: 141 ↛ 142line 141 didn't jump to line 142 because the condition on line 141 was never true

142 ctx.log(f"{name}: not enough input activity.") 

143 continue 

144 # Ratio of total achieved to total desired magnitude: < 1 means the vehicle falls short of the command. 

145 ratio = sum(abs(achieved) for _, achieved in pairs) / sum(abs(desired) for desired, _ in pairs) 

146 # Root-mean-square tracking error, relative to full scale. 

147 rms_error = math.sqrt(sum((desired - achieved) ** 2 for desired, achieved in pairs) / len(pairs)) / full_scale 

148 ctx.log( 

149 f"{name}: achieved/desired = {ratio:.2f}, RMS error {rms_error:.0%} of full scale " 

150 "(ratio < 0.9 -> raise I/FF, jerky or overshooting -> lower P). Edit values above and repeat." 

151 )

205def action_verify_speed_controller(ctx): 

206 """Guided-mode speed steps; refine ATC_SPEED_P/I automatically (or a manual Acro tracking report).""" 

207 if ctx.value("how_verify", "auto") == "manual":

211 target = ctx.number("v_step", DEFAULT_SPEED_STEP) 

212 _ensure_guided_can_reach(ctx, target) 

213 

214 # Hold the step as long as the usable runway allows at the target speed (0.1 m/s floor avoids dividing by 0).
