1"""rover_wizard.motion.step_tuning - tuning a controller (P, I and the steering's FF) from step responses. 

2 

3Each round the vehicle follows a step of its target (speed or turn rate); step_response_metrics() measures how it 

4followed, GainTuner.next_gains() decides the gains for the next round. Pure functions and state only: no vehicle 

5access, no UI, no clock. 

6 

7What makes it robust against real data: 

8* The target is what ArduPilot actually aimed for, when its controller reports it (PID_TUNING): ArduPilot limits the 

9 requested target first (ATC_STR_ACC_MAX, ATC_STR_RAT_MAX, ATC_TURN_MAX_G, ATC_ACCEL_MAX ...), and no gain can make 

10 the vehicle follow more than that. 

11* Overshoot is read from the smoothed signal, and overshoot or a steady-state error within the measurement noise 

12 counts as none: a gyro on a bumping vehicle is noisy, and single spikes are not the controller's doing. 

13* While the output is at its limit (full steering lock, full throttle) the target can't be reached at all: the 

14 gains are left alone. 

15* A steady-state error is corrected through FF where it can be (the output the vehicle needed, per unit achieved): 

16 ArduPilot's rate controllers rely on FF for it, I only builds up slowly within a short step. 

17* When measurement noise makes the controller's output jitter, an error filter (FLTE) well above the loop's 

18 bandwidth is proposed (suggest_error_filter). 

19* A gain whose change reverses the previous round's direction changes only half as much (in proportion) as before, 

20 so noisy rounds don't make the gains swing up and down; tuning ends once the changes have become small. 

21"""

29# fewer samples inside a step than this -> no metrics 

30STEP_MIN_SAMPLES = 8 

31# "t63": time to reach 63 % of the target (one time constant) 

32STEP_RISE_FRACTION = 0.63 

33# steady-state value = mean over the last ≤ 2.5 s of the step ... 

34STEP_TAIL_MAX_SECONDS = 2.5 

35# ... or the last half of the step, whichever is shorter (long enough to average a noisy gyro) 

36STEP_TAIL_FRACTION = 0.5 

37# [s] the signal is averaged over this window (centred) before its peak and rise time are read 

38SMOOTHING_SECONDS = 0.3 

39# a deviation smaller than this many standard deviations of its measurement counts as noise 

40NOISE_SIGMAS = 2.0 

41# |output| at or above this (ArduPilot's controller outputs are -1..1): the controller is at its limit ... 

42SATURATED_OUTPUT = 0.95 

43# ... for more than this share of the steady part of the step 

44SATURATED_SHARE = 0.5 

45 

46# rules (first match wins): thresholds on the step response, and (P, I) multipliers to apply. 

47# overshoot above 20 %: far too aggressive 

48LARGE_OVERSHOOT = 0.20 

49# overshoot of 10..20 %: slightly too aggressive 

50SMALL_OVERSHOOT = 0.10 

51# steady-state error above 10 %: FF or integrator too weak 

52LARGE_STEADY_ERROR = 0.10 

53# steady-state error of 5..10 % 

54SMALL_STEADY_ERROR = 0.05 

55# lower P and I clearly 

56LARGE_OVERSHOOT_FACTORS = (0.75, 0.6) 

57# lower P and I slightly 

58SMALL_OVERSHOOT_FACTORS = (0.85, 0.8) 

59# more proportional action 

60SLOW_RESPONSE_FACTORS = (1.3, 1.0) 

61# more integral action (and a little P to help) - when there is no FF to correct 

62LARGE_ERROR_FACTORS = (1.1, 1.5) 

63# a little more integral action 

64SMALL_ERROR_FACTORS = (1.0, 1.25) 

65# an FF correction changes FF by at most this factor (either way) per round 

66FF_MAX_FACTOR = 2.0 

67# each reversal of a gain's direction shrinks its changes (in log space, i.e. the exponent of its factor) by this 

68REVERSAL_DAMPING = 0.5 

69# [s] the P and D parts' jitter is measured around their average over this window 

70JITTER_WINDOW_SECONDS = 0.5 

71# the output may jitter by this much (one standard deviation, of full output) before an error filter is proposed 

72JITTER_MAX = 0.05 

73# an error filter this many times above the control loop's bandwidth hardly delays the control 

74FILTER_BANDWIDTH_FACTOR = 5 

75# [Hz] never proposed lower 

76ERROR_FILTER_MIN_HZ = 1.0 

77# a proposal must be below this fraction of the current filter to count 

78FILTER_CLEARLY_LOWER = 0.9 

79# changes of every gain within ±3 %: tuning has converged 

80CONVERGED_CHANGE = 0.03

124def centred_average(times: list, values: list, window: float) -> list: 

125 """For every sample: the mean of the samples within ±window/2 seconds around it.""" 

126 averages = [] 

127 start = end = 0 

128 running = 0.0 

129 for index, time in enumerate(times): 

130 # extend the window to the right, then drop samples on the left 

131 while end < len(times) and times[end] <= time + window / 2: 

132 running += values[end] 

133 end += 1 

134 while times[start] < time - window / 2: 

135 running -= values[start] 

136 start += 1 

137 averages.append(running / (end - start)) 

138 return averages

141def step_response_metrics( 

142 times: list, 

143 values: list, 

144 target: float, 

145 step_on: float, 

146 step_off: float, 

147 desired: Optional[list] = None, 

148 outputs: Optional[list] = None, 

149 feedback: Optional[list] = None, 

150) -> Optional[StepResponse]: 

151 """Overshoot, steady-state error and rise time of `values` following a step to `target` between the two times. 

152 

153 `desired` (ArduPilot's own target per sample), `outputs` (its controller output per sample) and `feedback` (the 

154 output's P + D part per sample) are optional, entries may be None. 

155 """ 

156 # Mirror negative steps so the maths below only has to handle a positive target. 

157 direction = 1 if target > 0 else -1 

158 count = len(times) 

159 desired = desired or [None] * count 

160 outputs = outputs or [None] * count 

161 segment = [ 

162 (time, value * direction, None if aim is None else aim * direction, None if out is None else out * direction) 

163 for time, value, aim, out in zip(times, values, desired, outputs) 

164 if step_on <= time <= step_off 

165 ] 

166 if len(segment) < STEP_MIN_SAMPLES or target == 0: 166 ↛ 167line 166 didn't jump to line 167 because the condition on line 166 was never true

167 return None 

168 

169 # The steady part = the last 1.5 s or last 30 % of the step, whichever is shorter. 

170 tail_start = step_off - min(STEP_TAIL_MAX_SECONDS, STEP_TAIL_FRACTION * (step_off - step_on)) 

171 tail = [entry for entry in segment if entry[0] >= tail_start] 

172 if len(tail) < 2: 

173 # the run ended (e.g. at the distance limit) before the end of the step: no steady state to judge 

174 return None 

175 

176 aims = [aim for _, _, aim, _ in tail if aim is not None] 

177 aimed = sum(aims) / len(aims) if aims else abs(target) 

178 if aimed <= 0: 178 ↛ 179line 178 didn't jump to line 179 because the condition on line 178 was never true

179 return None 

180 

181 segment_times = [time for time, _, _, _ in segment] 

182 raw = [value for _, value, _, _ in segment] 

183 smoothed = centred_average(segment_times, raw, SMOOTHING_SECONDS) 

184 # noise = what smoothing removed (its standard deviation), from the steady part 

185 residuals = [value - average for (time, value, _, _), average in zip(segment, smoothed) if time >= tail_start] 

186 noise = math.sqrt(sum(residual**2 for residual in residuals) / len(residuals)) 

187 per_window = max(1, len(segment) * SMOOTHING_SECONDS / max(step_off - step_on, SMOOTHING_SECONDS)) 

188 # The highest of n independent noise values lies about sqrt(2 ln n) standard deviations above their mean: noise 

189 # alone makes the smoothed peak that much higher (n = independent windows in the step), plus NOISE_SIGMAS. 

190 windows = max(2.0, (step_off - step_on) / SMOOTHING_SECONDS) 

191 peak_band = (math.sqrt(2 * math.log(windows)) + NOISE_SIGMAS) * noise / math.sqrt(per_window) 

192 

193 final_value = sum(value for _, value, _, _ in tail) / len(tail) 

194 # Overshoot = how far the smoothed peak went past the target - or past the final value if the vehicle settled 

195 # above the target (that is a steady-state error, not a controller ringing) - unless within the smoothed noise. 

196 excess = max(smoothed) - max(aimed, final_value) 

197 overshoot = excess / aimed if excess > peak_band else 0.0 

198 # Steady-state error = fraction of the target missing at the end, unless within the noise of the mean. 

199 missing = aimed - final_value 

200 measured_error = missing / aimed 

201 steady_state_error = measured_error if abs(missing) > NOISE_SIGMAS * noise / math.sqrt(len(tail)) else 0.0 

202 # Rise time = seconds until the smoothed value first reached 63 % of the target (one first-order time constant). 

203 rise_time = next( 

204 (time - step_on for time, value in zip(segment_times, smoothed) if value >= STEP_RISE_FRACTION * aimed), None 

205 ) 

206 tail_outputs = [out for _, _, _, out in tail if out is not None] 

207 saturated = bool(tail_outputs) and ( 

208 sum(abs(out) >= SATURATED_OUTPUT for out in tail_outputs) > SATURATED_SHARE * len(tail_outputs) 

209 ) 

210 

211 return StepResponse( 

212 overshoot=overshoot, 

213 steady_state_error=steady_state_error, 

214 rise_time_63=rise_time, 

215 final_value=final_value, 

216 target=aimed, 

217 noise=noise, 

218 measured_error=measured_error, 

219 saturated=saturated, 

220 output=sum(tail_outputs) / len(tail_outputs) if tail_outputs else None, 

221 jitter=_jitter(times, feedback, tail_start, step_off) if feedback else None, 

222 )

225def _jitter(times: list, feedback: list, start: float, end: float) -> Optional[float]: 

226 """Standard deviation of the feedback parts around their own (centred) average between start and end.""" 

227 pairs = [] 

228 for time, part in zip(times, feedback): 

229 # each report once: samples come faster than the reports, and a repeated one would hide the jitter 

230 if part is not None and start <= time <= end and (not pairs or part != pairs[-1][1]): 

231 pairs.append((time, part)) 

232 if len(pairs) < STEP_MIN_SAMPLES: 

233 return None 

234 pair_times = [time for time, _ in pairs] 

235 parts = [part for _, part in pairs] 

236 averages = centred_average(pair_times, parts, JITTER_WINDOW_SECONDS) 

237 return math.sqrt(sum((part - average) ** 2 for part, average in zip(parts, averages)) / len(parts))

240def suggest_error_filter(response: Optional[StepResponse], p_gain: float, current_hz: float) -> tuple: 

241 """(error filter frequency [Hz] to propose or None, why) when noise makes the controller's output jitter. 

242 

243 The jitter is the controller's own (from its reports), else estimated as P x measurement noise. Above 

244 JITTER_MAX, an error filter (FLTE) at FILTER_BANDWIDTH_FACTOR times the loop's bandwidth (from the measured rise 

245 time: 1 / (2 pi t63)) takes out the noise while delaying the control hardly at all - a lower filter would add 

246 lag the loop notices. Proposed only if it is clearly below the current filter. 

247 """ 

248 if response is None or response.rise_time_63 is None: 

249 return None, "" 

250 jitter = response.jitter if response.jitter is not None else p_gain * response.noise 

251 if jitter <= JITTER_MAX: 

252 return None, "" 

253 bandwidth = 1 / (2 * math.pi * response.rise_time_63) 

254 proposal = max(ERROR_FILTER_MIN_HZ, round(FILTER_BANDWIDTH_FACTOR * bandwidth, 1)) 

255 shown = f"the output jitters by ±{jitter:.0%} from measurement noise" 

256 if current_hz and proposal >= current_hz * FILTER_CLEARLY_LOWER: 

257 return None, f"{shown}, and the error filter is already at {current_hz:g} Hz: lower P, or reduce vibration" 

258 return proposal, f"{shown} -> error filter {proposal:g} Hz"

261class GainTuner: 

262 """Decides a controller's gains round by round from its step responses (see the module docstring).""" 

263 

264 def __init__(self, gains: dict, slow_rise_time: float) -> None: 

265 """Start from the current gains ("p", "i" and, if FF is to be corrected, "ff"); a response slower than 

266 slow_rise_time [s] to 63 % counts as too slow.""" 

267 self.gains = dict(gains) 

268 self.slow_rise_time = slow_rise_time 

269 # gain -> direction of its last change (+1 up, -1 down) and how often its direction reversed 

270 self._last_direction: dict = {} 

271 self._reversals: dict = {} 

272 

273 def next_gains(self, response: Optional[StepResponse]) -> GainChange: 

274 """The gains for the next round (also kept as self.gains).""" 

275 if response is None: 

276 return GainChange(dict(self.gains), "no usable data", blocked=True) 

277 if response.saturated: 

278 return GainChange( 

279 dict(self.gains), 

280 "the output was at its limit (full steering lock / full throttle): the target can't be reached, " 

281 "whatever the gains - lower the step target", 

282 blocked=True, 

283 ) 

284 

285 if response.overshoot > LARGE_OVERSHOOT: 

286 return self._scale(LARGE_OVERSHOOT_FACTORS, f"overshoot {response.overshoot:.0%} -> lower P and I") 

287 if response.overshoot > SMALL_OVERSHOOT: 

288 return self._scale(SMALL_OVERSHOOT_FACTORS, f"overshoot {response.overshoot:.0%} -> slightly lower P and I") 

289 error = response.steady_state_error 

290 # FF first: with too little FF the vehicle also gets to the target late, which more P wouldn't fix. The 

291 # correction measures what FF should be, so it may act on an error within the noise: a right FF changes 

292 # little (and noise-driven back-and-forth is damped), which ends tuning. 

293 has_ff = "ff" in self.gains and response.output is not None and response.final_value > 0 

294 if has_ff and abs(response.measured_error) > SMALL_STEADY_ERROR: 

295 return self._correct_ff(response) 

296 # Never reached 63 % of the target, or too late: not enough proportional action. 

297 if response.rise_time_63 is None or response.rise_time_63 > self.slow_rise_time: 

298 return self._scale(SLOW_RESPONSE_FACTORS, "response too slow -> raise P") 

299 

300 if abs(error) > SMALL_STEADY_ERROR: 

301 factors = LARGE_ERROR_FACTORS if abs(error) > LARGE_STEADY_ERROR else SMALL_ERROR_FACTORS 

302 more = "raise I" if abs(error) > LARGE_STEADY_ERROR else "slightly raise I" 

303 return self._scale(factors, f"steady-state error {error:.0%} -> {more}") 

304 

305 return GainChange(dict(self.gains), "tracking is good", converged=True)

307 def _correct_ff(self, response: StepResponse) -> GainChange: 

308 """FF = the output the vehicle needed per unit it achieved, so FF alone delivers it next time.""" 

309 old = self.gains["ff"] 

310 needed = response.output / response.final_value 

311 new = max(old / FF_MAX_FACTOR, min(old * FF_MAX_FACTOR, needed)) if old > 0 else needed 

312 return self._apply( 

313 {"ff": new / old if old > 0 else 1.0}, 

314 f"steady-state error {response.measured_error:.0%} -> FF {old:.3g} -> {new:.3g} " 

315 "(the output the turn needed)", 

316 absolute={"ff": new} if old <= 0 else None, 

317 )

319 def _scale(self, factors: tuple, reason: str) -> GainChange: 

320 """Multiply P and I by the rule's factors (damped, see _apply).""" 

321 p_factor, i_factor = factors 

322 return self._apply({"p": p_factor, "i": i_factor}, reason)

324 def _apply(self, factors: dict, reason: str, absolute: Optional[dict] = None) -> GainChange: 

325 """Apply multipliers to the gains; a gain whose direction reverses changes less from now on (in log space, 

326 halved with every reversal). Converged once every change is within CONVERGED_CHANGE.""" 

327 largest_change = 0.0 

328 for name, factor in factors.items(): 

329 if factor == 1.0: 

330 continue 

331 direction = 1 if factor > 1 else -1 

332 previous = self._last_direction.get(name) 

333 if previous is not None and previous != direction: 

334 self._reversals[name] = self._reversals.get(name, 0) + 1 

335 self._last_direction[name] = direction 

336 damped = factor ** (REVERSAL_DAMPING ** self._reversals.get(name, 0)) 

337 self.gains[name] *= damped 

338 largest_change = max(largest_change, abs(damped - 1)) 

339 for name, value in (absolute or {}).items(): 339 ↛ 340line 339 didn't jump to line 340 because the loop on line 339 never started

340 self.gains[name] = value 

341 largest_change = 1.0 

342 if largest_change < CONVERGED_CHANGE: 

343 return GainChange(dict(self.gains), reason + " - the changes have become small", converged=True) 

344 return GainChange(dict(self.gains), reason)
