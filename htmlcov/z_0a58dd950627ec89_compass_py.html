100def action_probe_compass_orientation(ctx): 

101 """Find COMPASS_ORIENTn of the compass used first (external) from two headings: facing magnetic north, then south. 

102 

103 With the orientation at None the compass reports in its own axes. The Earth's field points north and (in the 

104 northern hemisphere) down, so facing north the reading is F = H*forward + Z*down, facing south 

105 S = -H*forward + Z*down (both in sensor axes). Hence F - S is the vehicle's forward axis and F + S its down axis 

106 (up in the southern hemisphere, where Z < 0). The rotation mapping both onto the vehicle axes is the answer. 

107 """ 

108 slot = primary_compass_slot(ctx) 

109 orient = orientation_name(ctx) 

110 if not is_external(ctx, slot): 

111 raise VehicleError( 

112 "The compass used first is internal: it follows the board orientation, no " f"{orient} needed." 

113 ) 

114 

115 original = ctx.param_value(orient) 

116 if not ctx.confirm("Start the compass orientation probe (two reboots, about a minute)?"): 

117 ctx.log("Cancelled - no changes made.") 

118 return 

119 

120 northern = _is_northern_hemisphere(ctx) 

121 with restore_on_failure(ctx, {orient: original}): 

122 ctx.log("Resetting the compass orientation to None and rebooting ...") 

123 ctx.set_param(orient, 0) 

124 action_reboot_flight_controller(ctx) 

125 ctx.pause( 

126 "Place the vehicle LEVEL with its front pointing to MAGNETIC NORTH, away from metal, then press Continue." 

127 ) 

128 facing_north = _average_compass(ctx) 

129 ctx.pause("Now turn it around: front pointing to MAGNETIC SOUTH, still level, then press Continue.") 

130 facing_south = _average_compass(ctx) 

131 # Difference = 2H x forward axis; sum = 2Z x down axis (Z < 0 in the southern hemisphere: flip it). 

132 forward = tuple(n - s for n, s in zip(facing_north, facing_south)) 

133 down = tuple((n + s) * (1 if northern else -1) for n, s in zip(facing_north, facing_south)) 

134 # |F - S| / 2 = H 

135 horizontal = sum(c * c for c in forward) ** 0.5 / 2 

136 # |F + S| / 2 = |Z| 

137 vertical = sum(c * c for c in down) ** 0.5 / 2 

138 if horizontal <= 0 or vertical / horizontal < MIN_VERTICAL_TO_HORIZONTAL: 

139 raise VehicleError( 

140 "The magnetic field is almost horizontal here (near the magnetic equator) - the " 

141 "orientation can't be determined this way; use the rotate-by-hand calibration." 

142 ) 

143 

144 name, confidence = best_matching_rotation([n for n, _ in ORIENTATION_CHOICES], down, forward) 

145 if name is None or confidence < MIN_ORIENTATION_CONFIDENCE: 145 ↛ 146line 145 didn't jump to line 146 because the condition on line 145 was never true

146 raise VehicleError( 

147 f"No mounting fits the readings well (best {name}, {confidence:.0%}) - check " 

148 "the headings, keep away from metal, and try again." 

149 ) 

150 value = dict(ORIENTATION_CHOICES)[name] 

151 keep = ctx.confirm(f"Detected compass orientation '{name}' ({confidence:.0%}). Apply it and reboot?") 

152 ctx.set_param(orient, value if keep else original) 

153 action_reboot_flight_controller(ctx) 

154 

155 if keep: 

156 ctx.set_values(COMPASS_ORIENT=value) 

157 ctx.log("Compass orientation applied.") 

158 else: 

159 ctx.log("Discarded - restored the original setting.")

348# ---- compass-motor interference (CompassMot) ------------------------------------------------------------------------- 

349# Compass-motor interference compensation ("CompassMot"). 

350# 

351# Current through the motors, the ESC and the power wires adds a magnetic field that the compass sees on top of the 

352# Earth's. ArduPilot can subtract it: COMPASS_MOTCT = 2 ("Use Current") adds COMPASS_MOTn_X/Y/Z [mGauss/A] times the 

353# battery current to the compass reading (AP_Compass_Backend::correct_field). 

354# 

355# ArduPilot measures these values itself only on Copter (ArduCopter/compassmot.cpp, MAV_CMD_PREFLIGHT_CALIBRATION 

356# param6 = 1); Rover doesn't implement that command. So the wizard measures the same way: the motors run at a few power 

357# levels with the wheels lifted, and the change of the compass field per amp of extra battery current, negated, is the 

358# compensation. "Use Throttle" (COMPASS_MOTCT = 1) has no effect on Rover: only Copter feeds the throttle to the 

359# compass (Compass::set_throttle), so the current-based type is the one to use - which needs a current sensor. 

360 

361COMPASS_MOTCT_CHOICES = [ 

362 ("Disabled", COMPASS_MOTCT_DISABLED), 

363 ("Use current (measured below)", COMPASS_MOTCT_CURRENT), 

364 ("Use throttle (no effect on Rover)", COMPASS_MOTCT_THROTTLE), 

365] 

366 

367# [%] highest motor power of the measurement (the pilot can change it), and its limits 

368DEFAULT_COMPASSMOT_POWER_PCT = 50 

369COMPASSMOT_POWER_MIN_PCT = 10 

370COMPASSMOT_POWER_MAX_PCT = 100 

371# powered levels per motor, evenly up to the highest power - each between two motor-off levels 

372COMPASSMOT_POWER_LEVELS = 3 

373# [s] each level: let current and field settle, then average them 

374LEVEL_SETTLE_SECONDS = 1.5 

375LEVEL_MEASURE_SECONDS = 3.0 

376# [s] the motor test of a level runs this much longer than the level, so it never lapses in between 

377LEVEL_TEST_OVERLAP_SECONDS = 1.0 

378# [s] telemetry older than this is not used 

379TELEMETRY_FRESH_SECONDS = 1.0 

380 

381# [A] below this much extra current the field change is mostly compass noise (ArduPilot's own minimum: 3 A at full 

382# throttle; lifted wheels draw less, but the levels are averaged over seconds) 

383MIN_CURRENT_RISE_AMPS = 1.0 

384# COMPASS_MOT_* to 0.1 mGauss/A 

385COMPASS_MOT_DECIMALS = 1 

386# [%] interference above this share of the Earth's field (at the measured current) is worth fixing at the source; 

387# ArduPilot's guidance for CompassMot: below 30 % is acceptable 

388STRONG_INTERFERENCE_PCT = 30

426def fit_interference(runs: list) -> tuple: 

427 """Field change per amp (x, y, z) [mGauss/A] and the largest extra current [A] over all runs. 

428 

429 Each powered level is compared with the mean of the motor-off levels on either side, so a slow drift of the 

430 field or of the idle current cancels out; the slope through the origin is the least-squares fit of field change 

431 against current rise. (None, 0) without any current rise. 

432 """ 

433 products = [0.0 for _ in AXES] 

434 squares = 0.0 

435 largest_rise = 0.0 

436 for run in runs: 

437 for before, level, after in zip(run[0::2], run[1::2], run[2::2]): 

438 rise = level.amps - (before.amps + after.amps) / 2 

439 for axis in AXES: 

440 products[axis] += rise * (level.field[axis] - (before.field[axis] + after.field[axis]) / 2) 

441 squares += rise * rise 

442 largest_rise = max(largest_rise, rise) 

443 

444 if squares <= 0: 

445 return None, 0.0 

446 return tuple(product / squares for product in products), largest_rise

533def action_measure_compass_motor_interference(ctx): 

534 """Measure the compass field per amp of motor current and propose COMPASS_MOTCT = 2 with COMPASS_MOTn_X/Y/Z.""" 

535 if not measures_current(ctx): 

536 raise VehicleError( 

537 "The battery monitor reports no current - the compensation needs a current sensor " 

538 "(set it up on the Battery monitor step)." 

539 ) 

540 

541 max_percent = ctx.number("compassmot_power", DEFAULT_COMPASSMOT_POWER_PCT) 

542 if not COMPASSMOT_POWER_MIN_PCT <= max_percent <= COMPASSMOT_POWER_MAX_PCT: 

543 raise VehicleError(f"Motor power must be {COMPASSMOT_POWER_MIN_PCT}..{COMPASSMOT_POWER_MAX_PCT} %.") 

544 

545 motors = _motors(ctx) 

546 if not all(ctx.output_with_function(function) for _, function in motors): 

547 raise VehicleError("Assign the motor output(s) on the Motor output step first.") 

548 

549 if not ctx.confirm(ctx.prompt("compass.compassmot_start")): 

550 ctx.log("Cancelled - no changes made.") 

551 return 

552 

553 runs = _run_motors(ctx, motors, max_percent) 

554 per_amp, rise = fit_interference(runs) 

555 if per_amp is None or rise < MIN_CURRENT_RISE_AMPS: 

556 raise VehicleError( 

557 f"The motors drew only {rise:.1f} A more than at rest - too little to measure the " 

558 "interference reliably. Raise the motor power and try again." 

559 ) 

560 # the Earth's field: the mean of every motor-off level 

561 resting = [level.field for run in runs for level in run[0::2]] 

562 earth = _length(_mean_vector(resting)) 

563 strength = _length(per_amp) 

564 

565 share = strength * rise / earth * PERCENT if earth else 0.0 

566 ctx.log( 

567 f"Interference: {strength:.1f} mGauss per A - {share:.0f} % of the Earth's field at {rise:.1f} A " 

568 + ctx.prompt("compass.compassmot_load_note") 

569 ) 

570 if share > STRONG_INTERFERENCE_PCT: 570 ↛ 571line 570 didn't jump to line 571 because the condition on line 570 was never true

571 ctx.log( 

572 "That is strong: move the compass further from motors, ESC and battery wires (twist power-wire " 

573 "pairs) - compensation only corrects part of it." 

574 ) 

575 compensation = [round(-value, COMPASS_MOT_DECIMALS) for value in per_amp] 

576 ctx.set_values( 

577 COMPASS_MOTCT=COMPASS_MOTCT_CURRENT, mot_x=compensation[0], mot_y=compensation[1], mot_z=compensation[2] 

578 ) 

579 ctx.log( 

580 f"Proposed: COMPASS_MOTCT = 2 (use current), compensation {compensation} mGauss/A for "
