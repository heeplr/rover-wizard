11# U+2581..U+2588: eight bar heights

13# characters per channel column (fits two-digit channel numbers and the +/- buttons)

19def bar_for_fraction(fraction: float) -> str: 

20 """One bar character for a position 0.0 (lowest bar) .. 1.0 (highest bar); values outside are clamped.""" 

21 # scaled to the eight bar heights and clamped 

22 level = max(0, min(len(BAR_LEVELS) - 1, int(fraction * len(BAR_LEVELS)))) 

23 return BAR_LEVELS[level]

26def bar_for_pwm(pwm: int) -> str: 

27 """One bar character for a pulse width; blank if the channel/output isn't active.""" 

28 if not pwm or pwm >= UINT16_MAX: 

29 return " " 

30 

31 # position of the pulse inside the nominal range, 0.0 (1000 us) .. 1.0 (2000 us) 

32 return bar_for_fraction((pwm - RC_PWM_NOMINAL_MIN) / (RC_PWM_NOMINAL_MAX - RC_PWM_NOMINAL_MIN))

35def bar_for_signed(value: float, full_scale: float) -> str: 

36 """One bar character for a value -full_scale (lowest) .. +full_scale (highest); zero is half height.""" 

37 # -full_scale .. +full_scale -> 0.0 .. 1.0 

38 return bar_for_fraction((value + full_scale) / (2 * full_scale))

78# ---- compass tape -------------------------------------------------------------------------------------------------- 

79# characters in the tape; odd, so the heading sits exactly in the middle column 

80COMPASS_TAPE_WIDTH = 47 

81# degrees per character (must divide 360, so the tape wraps around north seamlessly) 

82DEGREES_PER_CELL = 3 

83# a "╷" every 10 degrees (on the nearest cell), a "." on the cells between 

84COMPASS_TICK_DEGREES = 10 

85COMPASS_TICK = "╷" 

86COMPASS_HALF_TICK = "." 

87COMPASS_INDICATOR = "▲" 

88FULL_CIRCLE_DEGREES = 360 

89COMPASS_POINTS = {0: "N", 45: "NE", 90: "E", 135: "SE", 180: "S", 225: "SW", 270: "W", 315: "NW"} 

90NO_HEADING_TEXT = "(no ATTITUDE data)" 

91CELLS_PER_CIRCLE = FULL_CIRCLE_DEGREES // DEGREES_PER_CELL

94def _nearest_cell(degrees: float) -> int: 

95 """The cell (0..CELLS_PER_CIRCLE-1) an angle is drawn on: the nearest one, so marks needn't fall on a cell.""" 

96 return round(degrees / DEGREES_PER_CELL) % CELLS_PER_CIRCLE 

97

99# the cells that show a tick, and the cells where a compass point starts 

100TICK_CELLS = {_nearest_cell(degrees) for degrees in range(0, FULL_CIRCLE_DEGREES, COMPASS_TICK_DEGREES)} 

101POINT_CELLS = {_nearest_cell(degrees): point for degrees, point in COMPASS_POINTS.items()}

104def compass_tape_text(heading: Optional[float]) -> str: 

105 """A scrolling compass tape with the heading [deg] in the middle, and a "▲" under it on the second line.""" 

106 if heading is None: 

107 return NO_HEADING_TEXT 

108 

109 half = COMPASS_TAPE_WIDTH // 2 

110 # the cell nearest to the heading sits in the middle column 

111 centre = _nearest_cell(heading) 

112 cells = [ 

113 COMPASS_TICK if (centre + column) % CELLS_PER_CIRCLE in TICK_CELLS else COMPASS_HALF_TICK 

114 for column in range(-half, half + 1) 

115 ] 

116 # Compass points start at their own cell; a two-letter point covers the cell after it. 

117 for column in range(-half, half + 1): 

118 point = POINT_CELLS.get((centre + column) % CELLS_PER_CIRCLE) 

119 for offset, letter in enumerate(point or ""): 

120 if column + offset <= half: 120 ↛ 119line 120 didn't jump to line 119 because the condition on line 120 was always true

121 cells[column + offset + half] = letter 

122 

123 return "".join(cells) + "\n" + " " * half + COMPASS_INDICATOR
