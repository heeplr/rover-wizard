15def action_suggest_navigation_params(ctx): 

16 """Propose WP_SPEED (cruise speed, at most 70 % of top speed) and WP_RADIUS (turn radius, >= 1 m, 0.5 m steps).""" 

17 top_speed = ctx.param_value("SPEED_MAX", 0) or 0 

18 cruise_speed = ctx.param_value("CRUISE_SPEED", 0) or 0 

19 turn_radius = ctx.param_value("TURN_RADIUS", 0) or 0 

20 waypoint_speed = cruise_speed if cruise_speed > 0 else WP_SPEED_FALLBACK_FRACTION * top_speed 

21 proposal = {} 

22 if waypoint_speed > 0: 

23 proposal["WP_SPEED"] = round( 

24 ( 

25 min(waypoint_speed, WP_SPEED_MAX_FRACTION * top_speed) 

26 if top_speed 

27 # to 0.1 m/s 

28 else waypoint_speed 

29 ), 

30 1, 

31 ) 

32 # Round to the nearest 0.5 m: x2, round, /2. 

33 proposal["WP_RADIUS"] = round(max(1.0, turn_radius) * 2) / 2 

34 ctx.set_values(**proposal) 

35 ctx.log( 

36 "Suggestions: WP_SPEED ~ cruise speed (<= 70% of top speed), WP_RADIUS ~ turn radius. Review and press Apply." 

37 )
