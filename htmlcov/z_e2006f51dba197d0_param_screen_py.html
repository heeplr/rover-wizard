66def restore_plan(current: dict, saved: dict) -> tuple: 

67 """What restoring `saved` (name -> value) onto the vehicle's `current` parameters does: (values to write, 

68 names already equal, differing names kept because they are read-only or NOT_RESTORED, names the vehicle lacks).""" 

69 writes, equal, kept, missing = {}, [], [], [] 

70 for name, value in sorted(saved.items()): 

71 if name not in current: 

72 missing.append(name) 

73 elif are_nearly_equal(current[name], value): 

74 equal.append(name) 

75 elif is_read_only(name) or name in NOT_RESTORED: 

76 kept.append(name) 

77 else: 

78 writes[name] = value 

79 

80 return writes, equal, kept, missing

88def restore_summary(path: str, plan: tuple) -> str: 

89 """What Restore will do with the file, for the confirmation and the log.""" 

90 writes, equal, kept, missing = plan 

91 lines = [f"{os.path.basename(path)}: {len(writes)} value(s) differ from the vehicle, {len(equal)} are equal."] 

92 if kept: 

93 lines.append(f"Not restored (read-only or managed by ArduPilot): {_names(kept)}") 

94 if missing:

97 "rebooting; restore again then." 

98 ) 

99 return "\n".join(lines)
