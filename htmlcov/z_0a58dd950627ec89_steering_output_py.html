1"""rover_wizard.steps.steering_output - the "Steering output" step.""" 

2 

3from __future__ import annotations 

4 

5from ..ardupilot_defs import SERVO_FUNCTION_GROUND_STEERING 

6from ..motion.servo_test import DEFAULT_STEERING_TEST_PERCENT, steering_test_action 

7from .util.outputs import OutputRole, assign_rover_outputs, build_servo_direction_fields 

8from ..model import Action, Step, value_field 

9 

10# proposed while no output steers yet: ArduPilot Rover's default layout has the steering servo on output 1 

11DEFAULT_STEERING_OUTPUT = 1 

12 

13STEERING_TYPE_CHOICES = [("Front-wheel steering (Ackermann)", "ackermann"), ("Skid steering (differential)", "skid")] 

14 

15STEERING_ROLE = OutputRole(SERVO_FUNCTION_GROUND_STEERING, "steering", "Steering servo", skid=False) 

16 

17 

18def apply_steering_output(ctx): 

19 """Save the steering type; front-wheel: assign GroundSteering to the chosen output, skid: remove it anywhere.""" 

20 steering_type = ctx.value("steer") 

21 ctx.store.set("steer", steering_type) 

22 ctx.log(f"Steering type: {steering_type}") 

23 wanted = {} if steering_type == "skid" else {int(ctx.number("ch_steer", 1)): SERVO_FUNCTION_GROUND_STEERING} 

24 assign_rover_outputs(ctx, wanted, (SERVO_FUNCTION_GROUND_STEERING,)) 

25 

26 

27def _is_front_wheel_selected(ctx) -> bool: 

28 """Front-wheel steering chosen in the Steering-type selector of this step (saved choice until it is shown). 

29 

30 The steering fields follow the selector itself, so switching the type shows/hides them at once, before Apply. 

31 """ 

32 selected = ctx.values.get("steer") 

33 return selected == "ackermann" if selected else not ctx.is_skid()
