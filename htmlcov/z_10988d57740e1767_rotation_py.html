1"""rover_wizard.util.rotation - board-orientation maths for the AHRS_ORIENTATION auto-probe. 

2 

3ArduPilot's AHRS_ORIENTATION names a rotation that turns vectors measured in the flight controller's own (sensor) 

4frame into the vehicle frame (x forward, y right, z down). This module builds those rotation matrices from the 

5names, and finds which name best explains two accelerometer readings taken in known poses. 

6 

7The per-axis formulas and the "roll first, then pitch, then yaw" order of compound names were checked against 

8ArduPilot's AP_Math implementation (e.g. Roll180 negates y and z; Yaw45 maps x -> (x-y)/sqrt2, y -> (x+y)/sqrt2). 

9"""

66def axis_rotation(axis: str, degrees: float) -> Matrix: 

67 """Right-handed rotation matrix for `degrees` about the vehicle's x (roll), y (pitch) or z (yaw) axis.""" 

68 radians = math.radians(degrees) 

69 cos_a = math.cos(radians) 

70 sin_a = math.sin(radians) 

71 # roll: y and z rotate, x stays 

72 if axis == "x": 

73 return ((1.0, 0.0, 0.0), (0.0, cos_a, -sin_a), (0.0, sin_a, cos_a)) 

74 # pitch: z and x rotate, y stays 

75 if axis == "y": 

76 return ((cos_a, 0.0, sin_a), (0.0, 1.0, 0.0), (-sin_a, 0.0, cos_a)) 

77 # yaw: x and y rotate, z stays 

78 if axis == "z": 78 ↛ 80line 78 didn't jump to line 80 because the condition on line 78 was always true

79 return ((cos_a, -sin_a, 0.0), (sin_a, cos_a, 0.0), (0.0, 0.0, 1.0)) 

80 raise ValueError(f"unknown rotation axis {axis!r}")

83def matrix_product(left: Matrix, right: Matrix) -> Matrix: 

84 """3x3 matrix product left @ right (apply `right` first, then `left`).""" 

85 # Element (row, col) = dot product of left's row with right's column. 

86 return tuple(tuple(sum(left[row][k] * right[k][col] for k in range(3)) for col in range(3)) for row in range(3))

89def transform(matrix: Matrix, vector: Vector) -> Vector: 

90 """3x3 matrix times a 3-vector.""" 

91 # Component i = dot product of the matrix's row i with the vector. 

92 return tuple(sum(matrix[row][k] * vector[k] for k in range(3)) for row in range(3))

96 """Transposed matrix - for a rotation matrix this is its inverse (rotates back).""" 

97 return tuple(zip(*matrix))

100def normalize(vector: Vector) -> Optional[Vector]: 

101 """Unit vector in the same direction, or None if the vector is too short to have a meaningful direction.""" 

102 # Euclidean length = sqrt(x² + y² + z²). 

103 length = math.sqrt(sum(component * component for component in vector)) 

104 # (almost) zero: no direction to speak of 

105 if length <= 1e-6: 105 ↛ 106line 105 didn't jump to line 106 because the condition on line 105 was never true

106 return None 

107 # Dividing every component by the length gives the same direction with length 1. 

108 return tuple(component / length for component in vector)

111def rotation_matrix_for_name(name: str) -> Matrix: 

112 """Rotation matrix (sensor frame -> vehicle frame) for a named orientation such as 'Yaw45Roll180'.""" 

113 # Parse "Roll<deg>", "Pitch<deg>", "Yaw<deg>" parts; a part that is absent means 0 deg ("None" has none). 

114 angles = {axis_name: int(degrees) for axis_name, degrees in _NAME_PART.findall(name)} 

115 matrix = IDENTITY 

116 # Compose in ArduPilot's order: roll is applied first, then pitch, then yaw (AP_Math/vector3.cpp). Pre-multiplying 

117 # each new rotation (new @ matrix) means it acts after the ones already in `matrix`. ArduPilot's names list them 

118 # as in the matrix product ("Yaw45Roll90" = yaw45 @ roll90: roll first) - the order in a name doesn't matter here. 

119 for axis_name, axis in (("Roll", "x"), ("Pitch", "y"), ("Yaw", "z")): 

120 if angles.get(axis_name): 

121 matrix = matrix_product(axis_rotation(axis, angles[axis_name]), matrix) 

122 return matrix

125def gravity_direction_from_accelerometer(acceleration: Vector) -> Vector: 

126 """Direction of gravity ("down") in the sensor frame, from a stationary accelerometer reading. 

127 

128 An accelerometer measures specific force, which at rest points *up* (a level ArduPilot board reads z ≈ -1 g). 

129 Gravity - the direction we want - is therefore the reading negated. 

130 """ 

131 return tuple(-component for component in acceleration)

134def best_matching_rotation(names: list, down_in_sensor_frame: Vector, forward_in_sensor_frame: Vector) -> tuple: 

135 """Which named rotation best maps the measured 'down' to vehicle-down and 'forward' to vehicle-forward. 

136 

137 `down_in_sensor_frame` is gravity's direction with the vehicle level; `forward_in_sensor_frame` is gravity's 

138 direction with the vehicle nose-down (then the vehicle's forward axis points at the ground). 

139 Returns (name, confidence 0..1); confidence 1.0 means both directions matched perfectly. 

140 """ 

141 down = normalize(down_in_sensor_frame) 

142 forward = normalize(forward_in_sensor_frame) 

143 if down is None or forward is None: 143 ↛ 144line 143 didn't jump to line 144 because the condition on line 143 was never true

144 return None, 0.0 

145 best_name = None 

146 best_score = -2.0 

147 for name in names: 

148 matrix = rotation_matrix_for_name(name) 

149 # Rotate both measured directions into the vehicle frame using this candidate. 

150 rotated_down = transform(matrix, down) 

151 rotated_forward = transform(matrix, forward) 

152 # Score = cos(angle to vehicle-down) + cos(angle to vehicle-forward). For unit vectors the cosine with 

153 # (0,0,1) is just the z component and with (1,0,0) the x component. Perfect match -> 1 + 1 = 2. 

154 score = rotated_down[2] + rotated_forward[0] 

155 if score > best_score: 

156 best_name = name 

157 best_score = score 

158 # Map the score range [-2, 2] onto a confidence in [0, 1] (negative scores are hopeless -> 0). 

159 return best_name, max(0.0, best_score / 2.0)
