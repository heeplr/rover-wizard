1"""rover_wizard.motion.servo_test - moving outputs by hand from the wizard (Test buttons, MIN/TRIM/MAX +/-, servo panel). 

2 

3How each kind of output can be moved, according to ArduPilot Rover's behaviour: 

4* Steering servo: Rover drives the steering output from the steering stick even while disarmed, and an RC override 

5 counts as stick input. So the wizard holds the steering "stick" (RC override, Manual mode) - no arming, no motor 

6 test. The override is released automatically a few seconds after the last click. 

7* Motor outputs: only ArduPilot's motor test drives them. The motor test ARMS the vehicle for its duration and 

8 DISARMS it afterwards (Rover/motor_test.cpp), needs calibrated RC and passes the arming checks. 

9* Outputs without a function (or RC pass-through / RCINx): MAV_CMD_DO_SET_SERVO moves them directly; ArduPilot 

10 refuses it for every other function ("Channel x is already in use"). 

11"""

103def arming_checks_skipped(vehicle): 

104 """Skip ArduPilot's arming checks while a motor test starts, then restore them. 

105 

106 The motor test arms the vehicle, and arming runs the arming checks - which a bench setup typically fails (no RC 

107 signal, compass not calibrated yet). Rover only checks when the test starts (Rover/motor_test.cpp), so the 

108 original value is restored right after the command - also when it fails. Written directly to the vehicle (not via 

109 the context), so this temporary change doesn't appear in the wizard's change log. 

110 """ 

111 if vehicle.has_parameter("ARMING_SKIPCHK"): 

112 name, skip_all = "ARMING_SKIPCHK", ARMING_SKIPCHK_ALL 

113 elif vehicle.has_parameter("ARMING_CHECK"): 113 ↛ 116line 113 didn't jump to line 116 because the condition on line 113 was always true

114 name, skip_all = "ARMING_CHECK", ARMING_CHECK_NONE 

115 else: 

116 yield 

117 return

120 if original == skip_all: 120 ↛ 121line 120 didn't jump to line 121 because the condition on line 120 was never true

121 yield 

122 return

124 vehicle.set_param(name, skip_all) 

125 try: 

126 yield 

127 finally: 

128 try: 

129 vehicle.set_param(name, original) 

130 except VehicleError as error:

170 # ------------------------------------------------------------------ steering (RC override, disarmed) 

171 def hold_steering(self, position: float, seconds: float = STEERING_HOLD_SECONDS) -> None: 

172 """Hold the steering stick at `position` (-1 left .. +1 right) for a few seconds; the vehicle stays disarmed.""" 

173 if self._ctx.vehicle.armed and not self._steering_driver.active: 

174 raise VehicleError("Disarm first: the steering test takes over the steering and throttle sticks.") 

175 

176 with self._lock: 

177 previous_mode = self._ctx.vehicle.custom_mode 

178 if not self._steering_driver.active: 

179 # Manual mode + override pump (throttle held at neutral) 

180 self._steering_driver.start() 

181 elif previous_mode != MODES["MANUAL"]: 

182 # e.g. a failsafe switched to Hold during the hold; the override keeps the radio failsafe cleared 

183 self._ctx.vehicle.set_mode("MANUAL") 

184 

185 if previous_mode != MODES["MANUAL"]: 

186 # Rover centres the steering in Hold (ModeHold::update) and other modes steer by themselves 

187 self._ctx.log( 

188 f"Switched to MANUAL mode for the steering test (was " 

189 f"{MODE_NAMES.get(previous_mode, previous_mode)})." 

190 ) 

191 

192 self._steering_position = _clamp(position, -1.0, 1.0) 

193 self._steering_driver.set(self._steering_position, 0.0) 

194 self._steering_release_at = timebase.now() + seconds 

195 self._steering_hold_seconds = seconds 

196 if self._release_thread is None: 

197 self._release_thread = threading.Thread(target=self._release_when_due, daemon=True) 

198 self._release_thread.start()

200 def _release_when_due(self) -> None: 

201 """Background: give the steering back to the transmitter once no click extended the hold.""" 

202 while True: 

203 with self._lock: 

204 # decided and done under the lock: a click can't slip in between 

205 if timebase.now() >= self._steering_release_at: 

206 was_held = self._steering_driver.active 

207 self._release_thread = None 

208 self._end_steering_hold() 

209 break 

210 timebase.sleep(RELEASE_CHECK_SECONDS) 

211 

212 if was_held: 

213 self._ctx.log(f"No button press for {self._steering_hold_seconds:g}s. Resetting servo output.")

233 def run_motor_test(self, instance: int, percent: float, seconds: float = MOTOR_TEST_SECONDS) -> None: 

234 """Run ArduPilot's motor test on one motor output; refuses if the pilot armed the vehicle.""" 

235 vehicle = self._ctx.vehicle 

236 if vehicle.armed and timebase.now() > self._motor_test_ends_at: 

237 raise VehicleError( 

238 "The vehicle is armed. ArduPilot's motor test disarms it when the test ends - " "disarm first."

241 with arming_checks_skipped(vehicle): 

242 result = vehicle.motor_test(instance, percent, seconds) 

243 

244 if result not in ACCEPTED_RESULTS: 

245 raise VehicleError(f"Motor test refused (result {result}): {MOTOR_TEST_REQUIREMENTS} {vehicle.last_text()}") 

246 # While our test runs the vehicle is armed by it; a further click just updates the running test. 

247 self._motor_test_ends_at = timebase.now() + seconds + 1.0

261 def nudge(self, output: int, direction: int) -> str: 

262 """Move one output a step up (+1) or down (-1); returns a line for the log.""" 

263 vehicle = self._ctx.vehicle 

264 function = vehicle.get_int(f"SERVO{output}_FUNCTION") 

265 if output in self._swapped or function in SERVO_FUNCTIONS_RESOLVED_AT_RUNTIME: 

266 return self._nudge_swapped(output, function, direction) 

267 

268 if function in SERVO_FUNCTIONS_DIRECTLY_SETTABLE: 

269 # Start from our last command, else what the output shows now, else its trim. 

270 if not self._pwm.get(output): 270 ↛ 272line 270 didn't jump to line 272 because the condition on line 270 was always true

271 self._pwm[output] = self._current_pulse(output) 

272 return f"SERVO{output}: {self._step_pulse(output, direction)} us" 

273 

274 if function == SERVO_FUNCTION_GROUND_STEERING: 

275 # Step from the held position if a hold is running, else from centre. 

276 start = self._steering_position if self._steering_driver.active else 0.0 

277 self.hold_steering(start + direction * PANEL_STEERING_STEP, PANEL_STEERING_HOLD_SECONDS) 

278 return ( 

279 f"{self._reported_pulse(output)} (steering held at {self._steering_position:+.0%}, " 

280 "disarmed, no motor test)" 

281 ) 

282 

283 if function in MOTOR_TEST_INSTANCE_BY_FUNCTION: 

284 # one step 

285 percent = _clamp(self._percent.get(output, 0) + direction * PANEL_MOTOR_STEP, -100, 100) 

286 instance = MOTOR_TEST_INSTANCE_BY_FUNCTION[function] 

287 self.run_motor_test(instance, percent) 

288 self._percent[output] = percent 

289 return ( 

290 f"{self._reported_pulse(output)} ({MOTOR_NAMES[instance]} motor test {percent:+.0f}% for " 

291 f"{MOTOR_TEST_SECONDS:g} s - arms the vehicle for the test, wheels off the ground!)" 

292 ) 

293 

294 raise VehicleError( 

295 f"SERVO{output} is used by function {function}: ArduPilot doesn't move it on command, and " 

296 "that function is only set up when the flight controller boots, so it can't be switched " 

297 "off for a moment either." 

298 )

300 # ------------------------------------------------------------------ function swap (e.g. gimbal servos) 

301 def _nudge_swapped(self, output: int, function: int, direction: int) -> str: 

302 """Move an output whose function ArduPilot resolves at runtime: switch its function to "none" (once), step 

303 the pulse with DO_SET_SERVO, and give the function back PANEL_STEERING_HOLD_SECONDS after the last click.""" 

304 vehicle = self._ctx.vehicle 

305 notes = [] 

306 if output not in self._swapped: 

307 # start where the function left the output 

308 self._pwm[output] = self._current_pulse(output) 

309 vehicle.set_param(f"SERVO{output}_FUNCTION", SERVO_FUNCTION_DISABLED) 

310 with self._lock: 

311 self._swapped[output] = function 

312 self._persist_swaps() 

313 notes.append(f"{SERVO_FUNCTIONS_RESOLVED_AT_RUNTIME.get(function, f'function {function}')} paused") 

314 # until ArduPilot has picked up the change, the function would overwrite our pulse 

315 timebase.sleep(FUNCTION_SWAP_SETTLE_SECONDS) 

316 

317 pwm = self._step_pulse(output, direction) 

318 with self._lock: 

319 self._swap_release_at = timebase.now() + PANEL_STEERING_HOLD_SECONDS 

320 if self._swap_thread is None: 

321 self._swap_thread = threading.Thread(target=self._restore_when_due, daemon=True) 

322 self._swap_thread.start() 

323 

324 notes.append(f"its function returns {PANEL_STEERING_HOLD_SECONDS:g} s after the last click") 

325 return f"SERVO{output}: {pwm} us ({', '.join(notes)})"

327 def _restore_when_due(self) -> None: 

328 """Background: give swapped outputs their function back once no click extended the hold.""" 

329 while True: 

330 with self._lock: 

331 # decided under the lock; a later click starts a new thread 

332 if timebase.now() >= self._swap_release_at: 

333 self._swap_thread = None 

334 # 0 = restore_functions() already ran (STOP, quit) 

335 already_restored = self._swap_release_at == 0.0 

336 break 

337 timebase.sleep(RELEASE_CHECK_SECONDS) 

338 

339 if not already_restored: 

340 self.restore_functions()

344 if not self._has_leftovers: 

345 return

354 def restore_functions(self) -> None: 

355 """Give every swapped output its original function back (a failed write is kept and retried later).""" 

356 with self._lock: 

357 self._swap_release_at = 0.0 

358 swapped = dict(self._swapped) 

359 

360 for output, function in swapped.items(): 

361 try: 

362 self._ctx.vehicle.set_param(f"SERVO{output}_FUNCTION", function) 

363 except VehicleError as error: 

364 self._ctx.log( 

365 f"❗ Could not give SERVO{output} its function {function} back ({error}) - the wizard " 

366 f"retries on its next start; or set SERVO{output}_FUNCTION = {function} yourself." 

367 ) 

368 continue 

369 

370 with self._lock: 

371 self._swapped.pop(output, None) 

372 self._pwm.pop(output, None) 

373 self._persist_swaps() 

374 name = SERVO_FUNCTIONS_RESOLVED_AT_RUNTIME.get(function, f"function {function}") 

375 self._ctx.log(f"SERVO{output}: back to {name}.")

402 def _reported_pulse(self, output: int) -> str: 

403 """'SERVOn: <pulse> us' as the output reports it once the command took effect. 

404 

405 For steering and motor outputs the wizard sends a stick position or a motor-test percentage, not a pulse 

406 width; ArduPilot turns that into the pulse (MIN/TRIM/MAX, reversal, ...), so the reported value is shown. 

407 """ 

408 vehicle = self._ctx.vehicle 

409 settled = timebase.now() + OUTPUT_SETTLE_SECONDS 

410 timebase.sleep(OUTPUT_SETTLE_SECONDS) 

411 # age <= time since `settled`: the latest report arrived after it (an earlier one may show the old pulse) 

412 while ( 

413 vehicle.age("SERVO_OUTPUT_RAW") > timebase.now() - settled 

414 and timebase.now() - settled < OUTPUT_REPORT_WAIT_SECONDS 

415 ): 

416 timebase.sleep(OUTPUT_REPORT_POLL_SECONDS) 

417 pulse = vehicle.servo_pulse(output) 

418 return f"SERVO{output}: {pulse} us" if pulse else f"SERVO{output}: (output not reported)"

420 # ------------------------------------------------------------------ MIN / TRIM / MAX buttons 

421 def nudge_endpoint(self, function: int, suffix: str, field_key: str, direction: int) -> None: 

422 """Change SERVOn_<suffix> by one step immediately and, for the steering servo, show the new position. 

423 

424 The steering servo is held at the changed end point (MIN = full left, MAX = full right, TRIM = centre, 

425 SERVOn_REVERSED respected). Motor outputs are not driven: a disarmed Rover sends the ESC its TRIM (no pulse 

426 if MOT_SAFE_DISARM = 1), and driving to MIN/MAX would mean full power. 

427 """ 

428 ctx = self._ctx 

429 output = ctx.output_with_function(function) 

430 if not output: 430 ↛ 431line 430 didn't jump to line 431 because the condition on line 430 was never true

431 raise VehicleError("Assign the output first (enter its number and press Apply).") 

432 

433 names = {part: f"SERVO{output}_{part}" for part in ("MIN", "TRIM", "MAX")} 

434 values = {part: ctx.param_value(name, RC_PWM_NOMINAL_TRIM) for part, name in names.items()} 

435 current = ctx.number(field_key, values[suffix]) 

436 # Keep MIN < TRIM < MAX: each end point may only move up to 1 us short of its neighbour. 

437 low = values["MIN"] + 1 if suffix == "TRIM" else values["TRIM"] + 1 if suffix == "MAX" else PWM_LIMITS[0] 

438 high = values["MAX"] - 1 if suffix == "TRIM" else values["TRIM"] - 1 if suffix == "MIN" else PWM_LIMITS[1] 

439 new = int(_clamp(current + direction * ENDPOINT_STEP, low, high)) 

440 ctx.set_param(names[suffix], new) 

441 ctx.set_values(**{field_key: new}) 

442 if function == SERVO_FUNCTION_GROUND_STEERING: 

443 reversed_output = ctx.param_int(f"SERVO{output}_REVERSED") == 1 

444 # Stick -1 drives the MIN end, +1 the MAX end; a reversed output swaps them. 

445 position = {"TRIM": 0.0, "MIN": -1.0, "MAX": 1.0}[suffix] * (-1 if reversed_output else 1) 

446 self.hold_steering(position) 

447 else: 

448 ctx.log( 

449 f"{names[suffix]} = {new} (motor outputs are not driven; while disarmed the ESC receives TRIM, " 

450 "or no pulse if MOT_SAFE_DISARM = 1)" 

451 )
